\section{Apêndice}

\begin{definition}[Ponto de máximo global]
	Seja $ f:A \subseteq \mathbb{R}^{n} \rightarrow \mathbb{R} $ uma função. Diz-se que $ P_{0} \in A $ é um ponto de máximo global (ou absoluto) de $ f $ se 
	
	\begin{equation*}
		f(P) \leq f(P_{0}), \quad \forall P \in A.
	\end{equation*}
	
	Nestas circunstâncias, diz-se que $ f(P_{0}) $ é o valor do máximo global (ou absoluto) da função $ f $ em $ A $. 
\end{definition}

De modo inteiramente análogo, 

\begin{definition}[Ponto de mínimo global]
	Seja $ f:A \subseteq \mathbb{R}^{n} \rightarrow \mathbb{R} $ uma função. Diz-se que $ P_{0} \in A$ é um ponto de mínimo global (ou absoluto) de $ f $ se
	
	\begin{equation*}
		f(P) \geq f(P_{0}), \quad \forall P \in A.
	\end{equation*}
	
	Neste caso, diz-se que $ f(P_{0}) $ é o valor mínimo global (ou absoluto) da função $ f $ e, $ A $
\end{definition}

Notoriamente, a função definida pode apresentar, para além dos extremantes globais (ou absolutos), extremantes locais, os quais são distinguidos entre pontos de máximo e mínimo locais. Assim, introduz-se as correspondentes definições 

\begin{definition}[Ponto de máximo local]
	Seja $ f: A \subseteq \mathbb{R}^{n} \rightarrow \mathbb{R} $ uma função. Diz-se que $ P_{0} \in A $ é ponto de máximo local (ou relativo) de $ f $ se existe uma bola aberta $ B_{\delta} \overset{\cdot}{=} B_{\delta}(P_{0})$, centrada em $ P_{0} $, de modo que:
	
	\begin{equation*}
		f(P) \leq f(P_{0}), \quad \forall P \in A \cap B_{\delta}.
	\end{equation*}
	
	Assim, diz-se que $ f(p_{0}) $ é o valor máximo local (ou relativo) da função $ f $
\end{definition}

De modo semelhante, tem-se que

\begin{definition}[Ponto de mínimo local]
	Seja $ f: A \subseteq \mathbb{R}^{n} \rightarrow \mathbb{R} $ uma função. Diz-se que $ P_{0} \in A $ é ponto de mínimo local (ou relativo) de $ f $ se existe uma bola aberta $ B_{\delta} \overset{\cdot}{=} B_{\delta}(P_{0})$, centrada em $ P_{0} $, de modo que:
	
	\begin{equation*}
		f(P) \geq f(P_{0}), \quad \forall P \in A \cap B_{\delta}.
	\end{equation*}
	
	Assim, diz-se que $ f(p_{0}) $ é o valor mínimo local (ou relativo) da função $ f $.
\end{definition}
